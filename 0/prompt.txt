  A simple registry to track accessibility-labeled buttons for screen readers. Needs a static map and clear method.

  import 'package:flutter/material.dart';

  class SemanticButtonRegistry {
    static final Map<String, GlobalKey> _registry = {};

    static void register(String label, GlobalKey key) {
      _registry[label] = key;
    }

    static GlobalKey? get(String label) => _registry[label];

    static void clear() => _registry.clear();
  }

  import 'package:flutter/material.dart';
  import 'semantic_button_registry.dart';

  class SemanticMaterialButton extends StatefulWidget {
    final String label;
    final VoidCallback? onPressed;
    final IconData icon;
    final Color? color;
    final String? tooltip;

    const SemanticMaterialButton({
      Key? key,
      required this.label,
      required this.icon,
      this.onPressed,
      this.color,
      this.tooltip,
    }) : super(key: key);

    @override
    State<SemanticMaterialButton> createState() => _SemanticMaterialButtonState();
  }

  class _SemanticMaterialButtonState extends State<SemanticMaterialButton> {
    final GlobalKey _key = GlobalKey();

    @override
    void initState() {
      super.initState();
      SemanticButtonRegistry.register(widget.label, _key);
    }

    @override
    Widget build(BuildContext context) {
      return Semantics(
        label: widget.label,
        button: true,
        enabled: widget.onPressed != null,
        child: Tooltip(
          message: widget.tooltip ?? widget.label,
          child: Material(
            key: _key,
            color: widget.color ?? Theme.of(context).colorScheme.primary,
            borderRadius: BorderRadius.circular(12),
            child: InkWell(
              borderRadius: BorderRadius.circular(12),
              onTap: widget.onPressed,
              child: Padding(
                padding: const EdgeInsets.all(12),
                child: Icon(widget.icon, color: Colors.white),
              ),
            ),
          ),
        ),
      );
    }
  }
